cmake_minimum_required(VERSION 3.22.1)
project(decode_test LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
foreach(option BUILD_CXXLIBS BUILD_PROGRAMS BUILD_EXAMPLES BUILD_TESTING BUILD_DOCS INSTALL_MANPAGES WITH_OGG BUILD_SHARED_LIBS INSTALL_CMAKE_CONFIG_MODULE)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
