cmake_minimum_required(VERSION 3.22)
project(local_music_tags LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_BINDINGS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WITH_ZLIB OFF CACHE BOOL "" FORCE)
set(WITH_APE OFF CACHE BOOL "" FORCE)
set(WITH_ASF OFF CACHE BOOL "" FORCE)
set(WITH_DSF OFF CACHE BOOL "" FORCE)
set(WITH_MATROSKA OFF CACHE BOOL "" FORCE)
set(WITH_MOD OFF CACHE BOOL "" FORCE)
set(WITH_SHORTEN OFF CACHE BOOL "" FORCE)
set(WITH_TRUEAUDIO OFF CACHE BOOL "" FORCE)
add_subdirectory(../.tools/taglib-2.3.2 taglib-build)
file(GLOB_RECURSE TAG_HEADERS "../.tools/taglib-2.3.2/taglib/*.h")
set(TAG_INCLUDE_DIRS "${CMAKE_CURRENT_BINARY_DIR}/taglib-build")
foreach(header IN LISTS TAG_HEADERS)
  get_filename_component(header_dir "${header}" DIRECTORY)
  list(APPEND TAG_INCLUDE_DIRS "${header_dir}")
endforeach()
list(REMOVE_DUPLICATES TAG_INCLUDE_DIRS)
if(ANDROID)
  add_library(localmusic_tags SHARED tags.cpp jni.cpp)
  target_include_directories(localmusic_tags PRIVATE ${TAG_INCLUDE_DIRS})
  target_link_libraries(localmusic_tags PRIVATE tag)
  target_link_options(localmusic_tags PRIVATE "-Wl,-z,max-page-size=16384")
  add_subdirectory(../native-usb usb-build)
else()
  add_executable(tag-check tags.cpp cli.cpp)
  target_include_directories(tag-check PRIVATE ${TAG_INCLUDE_DIRS})
  target_link_libraries(tag-check PRIVATE tag)
endif()
