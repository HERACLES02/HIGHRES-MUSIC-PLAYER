cmake_minimum_required(VERSION 3.22)
project(local_music_peer_flac LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_CXXLIBS OFF CACHE BOOL "" FORCE)
set(BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(INSTALL_MANPAGES OFF CACHE BOOL "" FORCE)
set(WITH_OGG OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../native-usb/vendor/flac flac-build EXCLUDE_FROM_ALL)
if(ANDROID)
    add_library(localmusic_peer_flac SHARED verify.cpp)
    target_link_libraries(localmusic_peer_flac PRIVATE FLAC)
    target_link_options(localmusic_peer_flac PRIVATE "-Wl,-z,max-page-size=16384")
else()
    add_executable(peer-flac-check verify.cpp)
    target_link_libraries(peer-flac-check PRIVATE FLAC)
endif()
